Capability Broker API

window.aiphone.* — the bridge between web apps and the OS

Every system capability is accessed through window.aiphone. This is the ONLY bridge. No other native APIs, no hidden channels, no escape hatches. Every call is mediated, enforced, and logged.

Core Pattern

// Every API call follows this flow:
// 1. App calls window.aiphone.capability.method()
// 2. Capability Broker checks app's manifest
// 3. Broker checks user's granted permissions
// 4. If approved: execute and log to audit trail
// 5. If denied: return error, log the attempt

const photo = await aiphone.camera.capture({ facing: 'back' });
// Audit: { app: "user.myapp", cap: "camera", act: "capture", result: "ok" }

Tier 0 — Storage

// Key-value (sandboxed per-app, encrypted)
await aiphone.storage.set('prefs', JSON.stringify({ theme: 'dark' }));
const prefs = await aiphone.storage.get('prefs');
await aiphone.storage.remove('prefs');
const keys = await aiphone.storage.list();
const usage = await aiphone.storage.usage();

// SQLite (structured data)
await aiphone.storage.execute(
  'CREATE TABLE IF NOT EXISTS items (id INTEGER PRIMARY KEY, name TEXT, done INTEGER)'
);
await aiphone.storage.execute(
  'INSERT INTO items (name, done) VALUES (?, ?)', ['Buy milk', 0]
);
const rows = await aiphone.storage.query(
  'SELECT * FROM items WHERE done = ?', [0]
);

Tier 1 — Camera, Microphone, Location

// Camera
const photo = await aiphone.camera.capture({ facing: 'back' });
const photos = await aiphone.camera.selectPhoto({ multiple: true });
const stream = await aiphone.camera.stream({ facing: 'front' });

// Microphone
const audio = await aiphone.microphone.record({ maxDuration: 30 });
const text = await aiphone.microphone.transcribe(audio); // OS-level STT

// Location
const pos = await aiphone.location.current();
// { lat: -37.81, lon: 144.96, accuracy: 5, altitude: 31 }
const handle = aiphone.location.watch(pos => console.log(pos));
const coords = await aiphone.location.geocode('Collins St, Melbourne');
const addr = await aiphone.location.reverseGeocode(-37.81, 144.96);

Tier 1 — Notifications, Sensors, AI

// Notifications
await aiphone.notifications.send('Reminder', 'Meeting in 10 min');
const id = await aiphone.notifications.schedule('Workout', 'Time to run', tomorrow9am);
aiphone.notifications.onAction((id, action) => handleAction(id, action));

// Sensors
aiphone.sensors.accelerometer((x, y, z) => updateTilt(x, y, z));
aiphone.sensors.stepCounter(steps => updateSteps(steps));
aiphone.sensors.barometer(pressure => updateAltitude(pressure));

// AI (OS-level LLM)
const answer = await aiphone.ai.ask('Summarize this receipt');
const cat = await aiphone.ai.classify('buy eggs', ['shopping','work','exercise']);
const data = await aiphone.ai.extract(
  'Meeting with John 3pm tomorrow at cafe',
  { person: 'string', time: 'string', place: 'string' }
);
const desc = await aiphone.ai.vision(photoBlob, 'What items are on this receipt?');
const vecs = await aiphone.ai.embeddings(['hello', 'world']);

Tier 2 — Filesystem, Contacts, Calendar

// Full filesystem access
const entries = await aiphone.filesystem.readDir('/sdcard/Documents');
const data = await aiphone.filesystem.readFile('/sdcard/Documents/notes.txt');
await aiphone.filesystem.writeFile('/sdcard/exports/data.json', jsonBlob);
aiphone.filesystem.watch('/sdcard/Downloads', event => onNewFile(event));

// Contacts
const contacts = await aiphone.contacts.list({ query: 'John' });
const id = await aiphone.contacts.create({ name: 'Jane', phone: '+61...' });
await aiphone.contacts.update(id, { email: 'jane@example.com' });

// Calendar
const events = await aiphone.calendar.events({ from: today, to: nextWeek });
await aiphone.calendar.create({
  title: 'Dentist', start: tomorrow2pm, end: tomorrow3pm,
  reminder: 60 // minutes before
});

Tier 2 — Bluetooth, NFC, USB, Network

// Bluetooth BLE
const devices = await aiphone.bluetooth.scan({ services: ['heart_rate'] });
const conn = await aiphone.bluetooth.connect(devices[0].id);
conn.subscribe('heart_rate', 'measurement', data => updateHR(data));

// BLE peripheral mode (phone becomes a BLE device)
await aiphone.bluetooth.advertise({
  name: 'My Sensor',
  services: [{ uuid: '...', characteristics: [...] }]
});

// NFC
const tag = await aiphone.nfc.read();
await aiphone.nfc.write({ records: [{ type: 'url', data: 'https://...' }] });

// USB host
const devices = await aiphone.usb.listDevices();
const conn = await aiphone.usb.connect(devices[0].id);

// Raw sockets
const tcp = await aiphone.network.tcp.connect('192.168.1.100', 8080);
tcp.send('hello');
tcp.onData(data => process(data));
const udp = await aiphone.network.udp.bind(9000);
udp.send('192.168.1.255', 9000, 'broadcast');

Tier 2 — Background, IPC

// Background tasks
await aiphone.background.register('sync', async () => {
  const data = await fetchNewData();
  await aiphone.storage.set('cache', JSON.stringify(data));
}, { every: '15m' });

// Inter-app communication
const result = await aiphone.ipc.send('user.calendar', {
  action: 'get_events', date: '2026-05-23'
});
aiphone.ipc.onMessage((from, msg) => handleMessage(from, msg));
aiphone.ipc.subscribe('data-updated', msg => refresh());

Tier 3 — Telephony, System

// Telephony (ALWAYS shows OS confirmation UI)
await aiphone.telephony.call('+61412345678');    // user must confirm
await aiphone.telephony.sendSMS('+61412345678', 'On my way'); // user must confirm
aiphone.telephony.onSMS((from, body) => logMessage(from, body));

// System
await aiphone.system.setBrightness(0.8);
await aiphone.system.setOrientation('landscape');
await aiphone.system.wakeLock(true);
const bat = await aiphone.system.batteryInfo(); // { level: 0.72, charging: true }
const net = await aiphone.system.networkInfo(); // { type: 'wifi', connected: true }
